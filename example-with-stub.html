Proper TCF 2.2 Stub Integration

1. Load Stub in <head> (CRITICAL)

<head>
    <!-- Load TCF API stub FIRST, before any other scripts -->
    <script src="./tcfapi-stub.min.js"></script>
    
    <!-- Other head content -->
    <title>Your Page</title>
</head>

2. Load Full CMP Later (Asynchronously)

<!-- Load the full CMP bundle anywhere after the stub -->
<script src="./dist/cmp.bundle.iife.js"></script>

3. Use __tcfapi Immediately

// This works immediately, even before the full CMP loads
__tcfapi('ping', 2, function(pingData, success) {
    console.log('CMP Status:', pingData.cmpStatus); // 'stub' initially, then 'loaded'
});

// Commands other than 'ping' are queued until full CMP loads
__tcfapi('addEventListener', 2, function(tcData, success) {
    console.log('TC Data:', tcData);
});
